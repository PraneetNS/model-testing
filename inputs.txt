# ─────────────────────────────────────────────────────────────────────────────
# 1. MODEL AUDITING & QUALITY GATES
# ─────────────────────────────────────────────────────────────────────────────

### Feature: Synchronous Quality Gate
# Endpoint: POST /api/v1/quality-gate
# Description: Upload model artifacts and datasets for an instant governance scan.
curl -X POST "http://localhost:8000/api/v1/quality-gate" \
     -H "Authorization: Bearer <TOKEN>" \
     -F "project_id=PROJ-123" \
     -F "model_file=@model.pkl" \
     -F "train_file=@training_data.csv" \
     -F "val_file=@validation_data.csv" \
     -F "target_column=churn" \
     -F "query=Check for accuracy and feature drift"

### Feature: CI/CD Deterministic Gate
# Endpoint: POST /api/v1/ci_cd
# Description: Same as above but returns HTTP 406 on failure to break pipelines.
curl -X POST "http://localhost:8000/api/v1/ci_cd" \
     -H "Authorization: Bearer <TOKEN>" \
     -F "project_id=CI-PROD-99" \
     -F "model_file=@model.pkl" \
     -F "train_file=@train.csv" \
     -F "val_file=@val.csv" \
     -F "target_column=label"

### Feature: Comprehensive Multi-Module Audit
# Endpoint: POST /api/v1/audit/run
# Description: Run performance, drift, fairness, and security checks.
curl -X POST "http://localhost:8000/api/v1/audit/run" \
     -H "Authorization: Bearer <TOKEN>" \
     -F "model_name=CreditRiskDetector" \
     -F "label_col=target" \
     -F "model_file=@model.joblib" \
     -F "val_file=@validation.csv" \
     -F "selected=performance" \
     -F "selected=drift" \
     -F "selected=fairness" \
     -F "selected=security"
